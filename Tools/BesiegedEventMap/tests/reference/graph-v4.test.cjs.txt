test('Governance cannot grant rewards without capture, on failure, twice, or without Giovanni',()=>{
  const p=data.preparation,s={captureConfirmed:true,governancePending:true,captureRoute:'varangians',giovanniPresent:true};
  assert.throws(()=>P.governance(p,{...s,captureConfirmed:false},'now'));
  assert.throws(()=>P.governance(p,{...s,captureRoute:'bribe'},'now'));
  assert.throws(()=>P.governance(p,{...s,giovanniPresent:false},'now'));
  const now=P.governance(p,s,'now');assert.equal(now.galataState,'giovanni');assert.equal(now.giovanniLoyaltyDelta,30);assert(now.defenceIncreasePending);
  assert.throws(()=>P.governance(p,now,'now'));
  const later=P.governance(p,s,'fief');assert.equal(later.galataState,'imperial');assert.equal(later.giovanniLoyaltyDelta,15);assert(later.galataPostwarPromise);assert(!later.defenceIncreasePending);
  assert.equal(P.governance(p,s,'neutral').galataState,'neutral');
});
